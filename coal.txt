; ---------------------------------------------------------------------
; Macros
; ---------------------------------------------------------------------
PrintMsg MACRO msgName          ; print a zero-terminated string
    push edx
    mov  edx, OFFSET msgName
    call WriteString
    pop  edx
ENDM

SetColor MACRO colorVal         ; change text color
    push eax
    mov  eax, colorVal
    call SetTextColor
    pop  eax
ENDM

PrintColored MACRO msgName, colorVal    ; print a string in a color, then reset to white
    SetColor colorVal
    PrintMsg msgName
    SetColor white
ENDM

ShowMenu MACRO titleMsg, menuMsg        ; clear screen, yellow title, menu text
    call Clrscr
    PrintColored titleMsg, yellow
    PrintMsg menuMsg
ENDM

Bar MACRO                       ; print a vertical border '|'
    push eax
    mov  al, '|'
    call WriteChar
    pop  eax
ENDM

CellNum MACRO value, wid        ; centred number in a bordered cell, then '|'
    mov  eax, value
    mov  ebx, wid
    call PrintCenterNum
ENDM

CellStr MACRO msgName, wid      ; centred string in a bordered cell, then '|'
    mov  edx, OFFSET msgName
    mov  ebx, wid
    call PrintCenterStr
ENDM

Border MACRO widArr, cnt        ; +-----+-----+ line for cnt columns
    push esi
    push ecx
    mov  esi, OFFSET widArr
    mov  ecx, cnt
    call PrintBorder
    pop  ecx
    pop  esi
ENDM

FillArr MACRO arr, val          ; set procCount DWORDs of arr to val
    mov  eax, val
    mov  edi, OFFSET arr
    mov  ecx, procCount
    rep  stosd
ENDM

CopyDwords MACRO src, dst, rowOff, cnt  ; copy cnt DWORDs of src into dst at byte offset rowOff
    mov  esi, OFFSET src
    lea  edi, dst[rowOff]
    mov  ecx, cnt
    rep  movsd
ENDM

ShiftLeft MACRO arr             ; move arr[EBX+1 .. ] one place left (EDX = element count)
    lea  esi, arr[ebx*4 + 4]
    lea  edi, arr[ebx*4]
    mov  ecx, edx
    rep  movsd
ENDM

; main: show the main menu and dispatch through menuTable until Exit.
main PROC
MainLoop:
    ShowMenu txtMainTitle, txtMainMenu
    INVOKE  ReadRange, ADDR prChoice, 1, MENU_EXIT
    cmp     eax, MENU_EXIT
    je      ExitProgram
    dec     eax                          ; choice -> table index
    call    DWORD PTR menuTable[eax*4]
    jmp     MainLoop
ExitProgram:
    PrintColored txtBye, lightGreen
    exit
main ENDP

; =====================================================================
;                         INPUT / OUTPUT HELPERS
; =====================================================================

; ReadRange: ask until a whole number in [minVal, maxVal] is entered.
; Returns EAX = value. (ReadInt sets the Overflow flag on bad input.)
ReadRange PROC USES edx, pPrompt:PTR BYTE, minVal:SDWORD, maxVal:SDWORD
RG_Retry:
    mov     edx, pPrompt
    call    WriteString
    call    ReadInt
    jo      RG_Bad
    cmp     eax, minVal
    jl      RG_Bad
    cmp     eax, maxVal
    jg      RG_Bad
    ret
RG_Bad:
    SetColor lightRed
    PrintMsg txtBad1
    mov     eax, minVal
    call    WriteDec
    PrintMsg txtBad2
    mov     eax, maxVal
    call    WriteDec
    PrintMsg txtBad3
    SetColor white
    jmp     RG_Retry
ReadRange ENDP

; ReadYesNo: read one key. Returns EAX = 1 for y/Y, otherwise 0.
ReadYesNo PROC
    call    ReadChar
    push    eax
    call    WriteChar                   ; echo the key
    call    Crlf
    pop     eax
    or      al, 20h                     ; force lower case
    cmp     al, 'y'
    sete    al
    movzx   eax, al
    ret
ReadYesNo ENDP

; NumDigits: EAX = unsigned value -> ECX = number of decimal digits.
NumDigits PROC USES eax ebx edx
    mov     ecx, 1
    mov     ebx, 10
ND_Loop:
    cmp     eax, 10
    jb      ND_Done
    xor     edx, edx
    div     ebx
    inc     ecx
    jmp     ND_Loop
ND_Done:
    ret
NumDigits ENDP

; PrintSpaces: print ECX spaces (nothing if ECX <= 0).
PrintSpaces PROC USES eax ecx
PS_Loop:
    cmp     ecx, 0
    jle     PS_Done
    mov     al, ' '
    call    WriteChar
    dec     ecx
    jmp     PS_Loop
PS_Done:
    ret
PrintSpaces ENDP

; PrintNumPad: print EAX, then pad with spaces to width EBX.
PrintNumPad PROC USES eax ebx ecx
    call    NumDigits
    call    WriteDec
    sub     ebx, ecx
    mov     ecx, ebx
    call    PrintSpaces
    ret
PrintNumPad ENDP

; StrLen: EDX = string -> ECX = length (REPNE SCASB).
StrLen PROC USES edi eax
    mov     edi, edx
    xor     eax, eax
    mov     ecx, -1
    cld
    repne   scasb
    not     ecx
    dec     ecx
    ret
StrLen ENDP

; PrintStrPad: print string EDX, then pad with spaces to width EBX.
PrintStrPad PROC USES eax ecx
    call    WriteString
    call    StrLen
    neg     ecx
    add     ecx, ebx
    call    PrintSpaces
    ret
PrintStrPad ENDP

; PrintFixed2: EAX = value x 100 -> prints "int.dd".
PrintFixed2 PROC USES eax ebx edx
    xor     edx, edx
    mov     ebx, 100
    div     ebx
    call    WriteDec                    ; integer part
    mov     al, '.'
    call    WriteChar
    mov     eax, edx                    ; fraction 0..99
    cmp     eax, 10
    jae     PF_Print
    push    eax
    mov     al, '0'                     ; leading zero for 0..9
    call    WriteChar
    pop     eax
PF_Print:
    call    WriteDec
    ret
PrintFixed2 ENDP

; PrintFixed2Pad: like PrintFixed2, padded to width EBX.
PrintFixed2Pad PROC USES eax ebx ecx edx
    push    eax
    push    ebx
    xor     edx, edx
    mov     ebx, 100
    div     ebx                         ; EAX = integer part
    pop     ebx
    call    NumDigits
    add     ecx, 3                      ; + ".dd"
    pop     eax
    call    PrintFixed2
    neg     ecx
    add     ecx, ebx
    call    PrintSpaces
    ret
PrintFixed2Pad ENDP

; PrintRepeat: print character AL, ECX times.
PrintRepeat PROC USES ecx
PR_Loop:
    cmp     ecx, 0
    jle     PR_Done
    call    WriteChar
    dec     ecx
    jmp     PR_Loop
PR_Done:
    ret
PrintRepeat ENDP

; PrintBorder: ESI = array of column widths, ECX = column count.
; Prints a line such as  +-------+-------+-------+
PrintBorder PROC USES eax ecx esi
    mov     al, '+'
    call    WriteChar
PB_Col:
    cmp     ecx, 0
    jle     PB_End
    push    ecx
    mov     ecx, [esi]
    mov     al, '-'
    call    PrintRepeat
    mov     al, '+'
    call    WriteChar
    pop     ecx
    add     esi, 4
    dec     ecx
    jmp     PB_Col
PB_End:
    call    Crlf
    ret
PrintBorder ENDP

; SplitPad: EBX = width, ECX = content length ->
;           ECX = left padding, EBX = right padding (never negative).
SplitPad PROC USES eax
    mov     eax, ebx
    sub     eax, ecx
    jge     SP_Ok
    xor     eax, eax
SP_Ok:
    mov     ebx, eax
    shr     eax, 1                      ; left = pad / 2
    sub     ebx, eax                    ; right = pad - left
    mov     ecx, eax
    ret
SplitPad ENDP

; PrintCenterStr: EDX = string, EBX = cell width.
; Prints the string centred in the cell followed by the '|' border.
PrintCenterStr PROC USES eax ebx ecx edx
    call    StrLen                      ; ECX = length
    call    SplitPad
    call    PrintSpaces
    call    WriteString
    mov     ecx, ebx
    call    PrintSpaces
    mov     al, '|'
    call    WriteChar
    ret
PrintCenterStr ENDP

; PrintCenterNum: EAX = unsigned number, EBX = cell width.
PrintCenterNum PROC USES eax ebx ecx edx
    push    eax
    call    NumDigits                   ; ECX = digits
    call    SplitPad
    call    PrintSpaces
    pop     eax
    call    WriteDec
    mov     ecx, ebx
    call    PrintSpaces
    mov     al, '|'
    call    WriteChar
    ret
PrintCenterNum ENDP

; PrintCenterPid: EAX = pid, EBX = cell width -> centred "P<pid>".
PrintCenterPid PROC USES eax ebx ecx edx
    push    eax
    call    NumDigits
    inc     ecx                         ; + the letter 'P'
    call    SplitPad
    call    PrintSpaces
    pop     eax
    call    PrintPid
    mov     ecx, ebx
    call    PrintSpaces
    mov     al, '|'
    call    WriteChar
    ret
PrintCenterPid ENDP

; PrintCenterFixed: EAX = value x100, EBX = cell width -> centred "int.dd".
PrintCenterFixed PROC USES eax ebx ecx edx
    push    eax
    push    ebx
    xor     edx, edx
    mov     ebx, 100
    div     ebx                         ; EAX = integer part
    call    NumDigits
    add     ecx, 3                      ; + ".dd"
    pop     ebx
    call    SplitPad
    call    PrintSpaces
    pop     eax
    call    PrintFixed2
    mov     ecx, ebx
    call    PrintSpaces
    mov     al, '|'
    call    WriteChar
    ret
PrintCenterFixed ENDP

; PrintSegCell: EAX = pid or IDLE_PID, EBX = cell width.
PrintSegCell PROC USES edx
    cmp     eax, IDLE_PID
    jne     PC_Proc
    mov     edx, OFFSET sIdle
    call    PrintCenterStr
    ret
PC_Proc:
    call    PrintCenterPid
    ret
PrintSegCell ENDP

; PrintPid: print "P<EAX>".
PrintPid PROC USES eax
    push    eax
    mov     al, 'P'
    call    WriteChar
    pop     eax
    call    WriteDec
    ret
PrintPid ENDP

; PrintAlgName: EAX = algorithm id (1..4) -> print its full name.
PrintAlgName PROC USES edx
    mov     edx, algNames[eax*4 - 4]
    call    WriteString
    ret
PrintAlgName ENDP

; ShowStat: print label EDX, then number EAX and a newline.
ShowStat PROC USES eax edx
    call    WriteString
    call    WriteDec
    call    Crlf
    ret
ShowStat ENDP

; Avg100: EAX = sum, EDI = count -> EAX = round(sum*100 / count).
; Returns 0 when count = 0.
Avg100 PROC USES ebx edx
    cmp     edi, 0
    je      AV_Zero
    mov     ebx, 100
    mul     ebx                         ; EDX:EAX = sum * 100
    mov     ebx, edi
    shr     ebx, 1                      ; + count/2 for rounding
    add     eax, ebx
    xor     edx, edx
    div     edi
    ret
AV_Zero:
    xor     eax, eax
    ret
Avg100 ENDP

; =====================================================================
;                          SCHEDULING ENGINE
; =====================================================================

; InitWorking: build a fresh working copy from the original data.
InitWorking PROC USES eax ecx esi edi
    cld
    mov     esi, OFFSET btArr           ; remaining = burst
    mov     edi, OFFSET remArr
    mov     ecx, procCount
    rep     movsd
    FillArr startArr, NOT_STARTED
    FillArr ctArr, 0
    FillArr doneArr, 0
    FillArr admitArr, 0
    mov     gCount, 0
    mov     segOverflow, 0
    mov     clockT, 0
    mov     rqHead, 0
    mov     rqTail, 0
    mov     rqCount, 0
    ret
InitWorking ENDP

; AddSegment: EAX = pid (or IDLE_PID), EBX = start, EDX = end.
; Extends the previous segment if same process and contiguous;
; ignores empty segments; flags overflow when the array is full.
AddSegment PROC USES eax ebx ecx edx esi
    cmp     edx, ebx
    jle     AS_Done
    mov     esi, gCount
    cmp     esi, 0
    je      AS_New
    cmp     gPid[esi*4 - 4], eax
    jne     AS_New
    cmp     gEnd[esi*4 - 4], ebx
    jne     AS_New
    mov     gEnd[esi*4 - 4], edx        ; merge with previous segment
    jmp     AS_Done
AS_New:
    cmp     esi, MAX_SEGMENTS
    jl      AS_Store
    mov     segOverflow, 1
    jmp     AS_Done
AS_Store:
    mov     gPid[esi*4], eax
    mov     gStart[esi*4], ebx
    mov     gEnd[esi*4], edx
    inc     gCount
AS_Done:
    ret
AddSegment ENDP

; SelectEarliest: EDI = flag array (doneArr / admitArr).
; Returns ESI = unflagged process with the earliest arrival
; (lowest index on ties), or -1.
SelectEarliest PROC USES eax ecx edx
    mov     esi, -1
    mov     edx, 7FFFFFFFh
    xor     ecx, ecx
SE_Loop:
    cmp     ecx, procCount
    jge     SE_Done
    cmp     DWORD PTR [edi+ecx*4], 0
    jne     SE_Next
    mov     eax, atArr[ecx*4]
    cmp     eax, edx
    jge     SE_Next
    mov     edx, eax
    mov     esi, ecx
SE_Next:
    inc     ecx
    jmp     SE_Loop
SE_Done:
    ret
SelectEarliest ENDP

; SelectMinReady: EAX = clock, EDI = key array, ECX = preferred index
; (-1 = none). Among arrived, unfinished processes picks the smallest
; key (lowest index on ties; the preferred process wins equal keys).
; Returns ESI = chosen index or -1.
SelectMinReady PROC USES eax ebx edx
    mov     esi, -1
    mov     edx, 7FFFFFFFh              ; best key so far
    xor     ebx, ebx
SM_Loop:
    cmp     ebx, procCount
    jge     SM_Pref
    cmp     doneArr[ebx*4], 0
    jne     SM_Next
    cmp     atArr[ebx*4], eax
    jg      SM_Next
    cmp     edx, [edi+ebx*4]
    jle     SM_Next
    mov     edx, [edi+ebx*4]
    mov     esi, ebx
SM_Next:
    inc     ebx
    jmp     SM_Loop
SM_Pref:                                ; tie-break in favour of ECX
    cmp     ecx, 0
    jl      SM_Done
    cmp     esi, -1
    je      SM_Done
    cmp     doneArr[ecx*4], 0
    jne     SM_Done
    cmp     atArr[ecx*4], eax
    jg      SM_Done
    cmp     edx, [edi+ecx*4]
    jne     SM_Done
    mov     esi, ecx
SM_Done:
    ret
SelectMinReady ENDP

; IdleToNext: EDI = flag array. Records an IDLE segment from the clock
; to the earliest unflagged arrival and advances the clock there.
; Returns ESI = that process, or -1 if none is left.
IdleToNext PROC USES eax ebx edx
    call    SelectEarliest
    cmp     esi, -1
    je      IN_Done
    mov     eax, IDLE_PID
    mov     ebx, clockT
    mov     edx, atArr[esi*4]
    call    AddSegment
    mov     clockT, edx
IN_Done:
    ret
IdleToNext ENDP

; ---------------------------------------------------------------------
; FCFS (non-preemptive): run in arrival order to completion
; ---------------------------------------------------------------------
RunFCFS PROC USES eax ebx ecx edx esi edi
    call    InitWorking
    mov     edi, OFFSET doneArr
    mov     ecx, procCount              ; processes still to schedule
FC_Loop:
    cmp     ecx, 0
    jle     FC_Done
    call    SelectEarliest              ; next process in arrival order
    mov     eax, atArr[esi*4]
    cmp     clockT, eax
    jge     FC_Run
    mov     ebx, clockT                 ; not arrived yet -> CPU idle
    mov     edx, eax
    mov     eax, IDLE_PID
    call    AddSegment
    mov     eax, atArr[esi*4]
    mov     clockT, eax
FC_Run:
    mov     eax, clockT
    mov     startArr[esi*4], eax
    mov     ebx, eax
    mov     edx, eax
    add     edx, btArr[esi*4]           ; runs until finished
    mov     eax, pidArr[esi*4]
    call    AddSegment
    mov     clockT, edx
    mov     ctArr[esi*4], edx
    mov     doneArr[esi*4], 1
    dec     ecx
    jmp     FC_Loop
FC_Done:
    ret
RunFCFS ENDP

; ---------------------------------------------------------------------
; Round Robin: circular ready queue (Enqueue / Dequeue)
; ---------------------------------------------------------------------

; Enqueue: ESI = process index -> add at the tail of the queue.
Enqueue PROC USES ebx
    mov     ebx, rqTail
    mov     rq[ebx*4], esi
    inc     ebx
    cmp     ebx, MAX_PROCESSES
    jl      EQ_Store
    xor     ebx, ebx                    ; wrap around
EQ_Store:
    mov     rqTail, ebx
    inc     rqCount
    ret
Enqueue ENDP

; Dequeue: remove the head of the queue -> ESI = process index.
Dequeue PROC USES ebx
    mov     ebx, rqHead
    mov     esi, rq[ebx*4]
    inc     ebx
    cmp     ebx, MAX_PROCESSES
    jl      DQ_Store
    xor     ebx, ebx                    ; wrap around
DQ_Store:
    mov     rqHead, ebx
    dec     rqCount
    ret
Dequeue ENDP

; AdmitArrivals: EAX = clock. Queue every not-yet-queued process with
; arrival <= clock, in arrival order.
AdmitArrivals PROC USES eax ebx edx esi edi
    mov     edx, eax
AA_Loop:
    mov     edi, OFFSET admitArr
    call    SelectEarliest
    cmp     esi, -1
    je      AA_Done
    cmp     atArr[esi*4], edx
    jg      AA_Done
    mov     admitArr[esi*4], 1
    call    Enqueue
    jmp     AA_Loop
AA_Done:
    ret
AdmitArrivals ENDP

RunRoundRobin PROC USES eax ebx ecx edx esi edi
    call    InitWorking
    mov     finishedCnt, 0
RB_Loop:
    mov     eax, finishedCnt
    cmp     eax, procCount
    jge     RB_Done
    mov     eax, clockT
    call    AdmitArrivals
    cmp     rqCount, 0
    jne     RB_Dispatch
    mov     edi, OFFSET admitArr        ; queue empty -> idle until next arrival
    call    IdleToNext
    cmp     esi, -1
    je      RB_Done
    jmp     RB_Loop
RB_Dispatch:
    call    Dequeue                     ; ESI = front process
    cmp     startArr[esi*4], NOT_STARTED
    jne     RB_Slice
    mov     eax, clockT
    mov     startArr[esi*4], eax        ; first CPU allocation
RB_Slice:
    mov     ecx, remArr[esi*4]          ; slice = min(remaining, quantum)
    cmp     ecx, rrQuantum
    jle     RB_Exec
    mov     ecx, rrQuantum
RB_Exec:
    mov     eax, pidArr[esi*4]
    mov     ebx, clockT
    mov     edx, ebx
    add     edx, ecx
    call    AddSegment
    mov     clockT, edx
    sub     remArr[esi*4], ecx
    mov     eax, clockT
    call    AdmitArrivals               ; arrivals queue BEFORE the requeue
    cmp     remArr[esi*4], 0
    jg      RB_Requeue
    mov     eax, clockT                 ; finished
    mov     ctArr[esi*4], eax
    mov     doneArr[esi*4], 1
    inc     finishedCnt
    jmp     RB_Loop
RB_Requeue:
    call    Enqueue                     ; unfinished -> back of the queue
    jmp     RB_Loop
RB_Done:
    ret
RunRoundRobin ENDP

; ---------------------------------------------------------------------
; Shared engine for Priority and SRTF: every time unit, re-select the
; ready process with the smallest key (so better arrivals preempt).
;   EDI = key array (priArr or remArr)
;   EBX = 1 -> running process wins equal keys
; ---------------------------------------------------------------------
RunTimeStepped PROC USES eax ebx ecx edx esi edi
    mov     preferFlag, ebx
    mov     metricBase, edi
    call    InitWorking
    mov     finishedCnt, 0
    mov     lastIdx, -1
TS_Loop:
    mov     eax, finishedCnt
    cmp     eax, procCount
    jge     TS_Done
    mov     eax, clockT
    mov     edi, metricBase
    mov     ecx, -1                     ; no preferred process by default
    cmp     preferFlag, 0
    je      TS_Select
    mov     ecx, lastIdx
TS_Select:
    call    SelectMinReady              ; ESI = process to run, or -1
    cmp     esi, -1
    jne     TS_Run
    mov     edi, OFFSET doneArr         ; nobody ready -> CPU idle
    call    IdleToNext
    cmp     esi, -1
    je      TS_Done
    mov     lastIdx, -1
    jmp     TS_Loop
TS_Run:
    cmp     startArr[esi*4], NOT_STARTED
    jne     TS_Exec
    mov     eax, clockT
    mov     startArr[esi*4], eax        ; record first start once
TS_Exec:
    mov     eax, pidArr[esi*4]          ; run exactly 1 time unit
    mov     ebx, clockT
    mov     edx, ebx
    inc     edx
    call    AddSegment
    mov     clockT, edx
    mov     lastIdx, esi
    dec     remArr[esi*4]
    jnz     TS_Loop                     ; not finished yet
    mov     eax, clockT
    mov     ctArr[esi*4], eax           ; finished at end of this unit
    mov     doneArr[esi*4], 1
    inc     finishedCnt
    mov     lastIdx, -1
    jmp     TS_Loop
TS_Done:
    ret
RunTimeStepped ENDP

; RunPriority: key = priority; equal priorities -> lowest input order.
RunPriority PROC USES ebx edi
    mov     edi, OFFSET priArr
    xor     ebx, ebx
    call    RunTimeStepped
    ret
RunPriority ENDP

; RunSRTF: key = remaining time; ties -> keep the running process.
RunSRTF PROC USES ebx edi
    mov     edi, OFFSET remArr
    mov     ebx, 1
    call    RunTimeStepped
    ret
RunSRTF ENDP

; AllocHistorySlot: returns EAX = slot for the next individual run.
; Ring buffer (slot = serial mod MAX_HISTORY); warns when it replaces
; the oldest run.
AllocHistorySlot PROC USES ebx ecx edx
    mov     eax, histSerial
    xor     edx, edx
    mov     ecx, MAX_HISTORY
    div     ecx
    mov     ebx, edx                    ; EBX = slot
    cmp     histSerial, MAX_HISTORY
    jb      AH_Fresh
    SetColor lightRed
    PrintMsg txtReplace
    mov     eax, slotSerial[ebx*4]
    call    WriteDec
    call    Crlf
    SetColor white
AH_Fresh:
    inc     histSerial
    mov     eax, histSerial
    mov     slotSerial[ebx*4], eax      ; run number shown to the user
    mov     lastSlot, ebx
    mov     eax, ebx
    ret
AllocHistorySlot ENDP

; SaveResult: EAX = slot. Copies settings, input, results and Gantt of
; the finished run into the slot (REP MOVSD per array).
SaveResult PROC USES eax ebx ecx edx esi edi
    mov     ebx, eax                    ; EBX = slot
    mov     edx, eax
    imul    edx, MAX_PROCESSES * 4      ; EDX = byte offset of process row
    imul    eax, MAX_SEGMENTS * 4       ; EAX = byte offset of segment row
    mov     ecx, curAlg
    mov     slotAlg[ebx*4], ecx
    xor     ecx, ecx                    ; quantum only matters for RR
    cmp     curAlg, ALG_RR
    jne     SR_Q
    mov     ecx, rrQuantum
SR_Q:
    mov     slotQuantum[ebx*4], ecx
    mov     ecx, procCount
    mov     slotN[ebx*4], ecx
    mov     ecx, gCount
    mov     slotSegCnt[ebx*4], ecx
    mov     ecx, segOverflow
    mov     slotOvf[ebx*4], ecx
    cld
    CopyDwords pidArr,   slotPid,    edx, procCount
    CopyDwords atArr,    slotAt,     edx, procCount
    CopyDwords btArr,    slotBt,     edx, procCount
    CopyDwords priArr,   slotPri,    edx, procCount
    CopyDwords ctArr,    slotCt,     edx, procCount
    CopyDwords startArr, slotStart,  edx, procCount
    CopyDwords gPid,     slotGPid,   eax, gCount
    CopyDwords gStart,   slotGStart, eax, gCount
    CopyDwords gEnd,     slotGEnd,   eax, gCount
    ret
SaveResult ENDP

; DisplayResult: EAX = slot. Header, Gantt chart, metrics, averages.
DisplayResult PROC USES eax esi
    mov     esi, eax

DR_NoQ:

DR_Metrics:

    ret
DisplayResult ENDP

; GenerateGanttChart: EAX = slot. Prints up to GANTT_PER_LINE cells per
; row: a label line (| P1 | P2 |) and a line of boundary times.
GenerateGanttChart PROC USES eax ebx ecx edx esi edi
    mov     esi, eax
    mov     ebx, esi
    imul    ebx, MAX_SEGMENTS           ; EBX = first segment of the slot
    mov     edi, slotSegCnt[esi*4]      ; EDI = number of segments
    xor     ecx, ecx                    ; ECX = first segment of this row
GG_Row:
    cmp     ecx, edi
    jge     GG_End
    mov     eax, ecx
    add     eax, GANTT_PER_LINE
    cmp     eax, edi
    jle     GG_SetEnd
    mov     eax, edi
GG_SetEnd:
    mov     chunkEnd, eax               ; row covers [ECX, chunkEnd)
    mov     chunkStart, ecx
    call    GanttBorder                 ; +------+------+
    Bar                                 ; |  P1  | IDLE |
    mov     edx, ecx
GG_L1:
    cmp     edx, chunkEnd
    jge     GG_L1End
    lea     eax, [ebx+edx]
    mov     eax, slotGPid[eax*4]
    push    ebx
    mov     ebx, GANTT_W
    call    PrintSegCell
    pop     ebx
    inc     edx
    jmp     GG_L1
GG_L1End:
    call    Crlf
    call    GanttBorder
    mov     edx, ecx
GG_L2:                                  ; times sit under each '+'
    cmp     edx, chunkEnd
    jge     GG_L2End
    lea     eax, [ebx+edx]
    mov     eax, slotGStart[eax*4]
    push    ebx
    mov     ebx, GANTT_W + 1
    call    PrintNumPad
    pop     ebx
    inc     edx
    jmp     GG_L2
GG_L2End:
    mov     eax, chunkEnd               ; end time of the last cell
    dec     eax
    add     eax, ebx
    mov     eax, slotGEnd[eax*4]
    call    WriteDec
    call    Crlf
    call    Crlf
    mov     ecx, chunkEnd
    jmp     GG_Row
GG_End:
    ret
GenerateGanttChart ENDP

; GanttBorder: border line for the cells chunkStart .. chunkEnd-1.
GanttBorder PROC USES ecx esi
    mov     ecx, chunkEnd
    sub     ecx, chunkStart
    mov     esi, OFFSET wGantt
    call    PrintBorder
    ret
GanttBorder ENDP

; CalculateMetrics: EAX = slot, ECX = process index.
; Sets mPid mAT mBT mPri mCT mTAT mWT mRT where
;   TAT = CT - AT,  WT = TAT - BT,  RT = first start - AT
CalculateMetrics PROC USES eax edx
    mov     edx, eax
    imul    edx, MAX_PROCESSES
    add     edx, ecx                    ; flat index (slot,i)
    mov     eax, slotPid[edx*4]
    mov     mPid, eax
    mov     eax, slotPri[edx*4]
    mov     mPri, eax
    mov     eax, slotAt[edx*4]
    mov     mAT, eax
    mov     eax, slotBt[edx*4]
    mov     mBT, eax
    mov     eax, slotCt[edx*4]
    mov     mCT, eax
    sub     eax, mAT
    mov     mTAT, eax
    sub     eax, mBT
    mov     mWT, eax
    mov     eax, slotStart[edx*4]
    sub     eax, mAT
    mov     mRT, eax
    ret
CalculateMetrics ENDP

; ComputeAverages: EAX = slot -> avgWT, avgTAT, avgRT (values x100).
ComputeAverages PROC USES eax ecx edx esi edi
    mov     esi, eax
    mov     sumWT, 0
    mov     sumTAT, 0
    mov     sumRT, 0
    mov     edi, slotN[esi*4]           ; EDI = n (Avg100 guards n = 0)
    xor     ecx, ecx
CV_Loop:
    cmp     ecx, edi
    jge     CV_Calc
    mov     eax, esi
    call    CalculateMetrics
    mov     eax, mWT
    add     sumWT, eax
    mov     eax, mTAT
    add     sumTAT, eax
    mov     eax, mRT
    add     sumRT, eax
    inc     ecx
    jmp     CV_Loop
CV_Calc:
    mov     eax, sumWT
    call    Avg100
    mov     avgWT, eax
    mov     eax, sumTAT
    call    Avg100
    mov     avgTAT, eax
    mov     eax, sumRT
    call    Avg100
    mov     avgRT, eax
    ret
ComputeAverages ENDP

; DisplayMetrics: EAX = slot. Per-process table plus the 3 averages.
DisplayMetrics PROC USES eax ebx ecx edx esi edi
    mov     esi, eax
    PrintColored hdrMetrics, yellow
    Border  wMetrics, 8
    SetColor yellow
    Bar
    CellStr hPID, TBL_W
    CellStr hAT,  TBL_W
    CellStr hBT,  TBL_W
    CellStr hPRI, TBL_W
    CellStr hCT,  TBL_W
    CellStr hTAT, TBL_W
    CellStr hWT,  TBL_W
    CellStr hRT,  TBL_W
    call    Crlf
    SetColor white
    Border  wMetrics, 8
    mov     edi, slotN[esi*4]
    xor     ecx, ecx
DM_Row:
    cmp     ecx, edi
    jge     DM_End
    mov     eax, esi
    call    CalculateMetrics
    Bar
    CellNum mPid, TBL_W
    CellNum mAT,  TBL_W
    CellNum mBT,  TBL_W
    CellNum mPri, TBL_W
    CellNum mCT,  TBL_W
    CellNum mTAT, TBL_W
    CellNum mWT,  TBL_W
    CellNum mRT,  TBL_W
    call    Crlf
    inc     ecx
    jmp     DM_Row
DM_End:
    Border  wMetrics, 8
DM_Avg:
    call    Crlf
    mov     eax, esi
    call    ComputeAverages
    PrintMsg txtAvgWT
    mov     eax, avgWT
    call    PrintFixed2
    call    Crlf
    PrintMsg txtAvgTAT
    mov     eax, avgTAT
    call    PrintFixed2
    call    Crlf
    PrintMsg txtAvgRT
    mov     eax, avgRT
    call    PrintFixed2
    call    Crlf
    ret
DisplayMetrics ENDP

; ViewPreviousResults: show the last run, an earlier run or the last comparison.
ViewPreviousResults PROC USES eax esi
VP_Loop:
    ShowMenu txtPrevTitle, txtPrevMenu
    INVOKE  ReadRange, ADDR prChoice, 1, 4
    cmp     eax, 4
    je      VP_Exit
    cmp     eax, 1
    je      VP_Last
    cmp     eax, 2
    je      VP_Earlier
    cmp     cmpValid, 0                 ; option 3: last comparison
    jne     VP_ShowCmp
    PrintMsg txtNoCmp
    jmp     VP_Wait
VP_ShowCmp:
    call    ShowComparison
    jmp     VP_Wait
VP_Last:
    cmp     histSerial, 0
    jne     VP_ShowLast
    PrintMsg txtNoRuns
    jmp     VP_Wait
VP_ShowLast:
    mov     eax, lastSlot
    call    DisplayResult
    jmp     VP_Wait
VP_Earlier:
    cmp     histSerial, 0
    jne     VP_Pick
    PrintMsg txtNoRuns
    jmp     VP_Wait
VP_Pick:
    call    PickRun                     ; ESI = slot or -1
    cmp     esi, -1
    je      VP_Wait
    mov     eax, esi
    call    DisplayResult
VP_Wait:
    call    Crlf
    call    WaitMsg
    jmp     VP_Loop
VP_Exit:
    ret
ViewPreviousResults ENDP

; ListHistory: print every stored individual run.
ListHistory PROC USES eax ecx
    PrintColored hdrRuns, yellow
    xor     ecx, ecx
LH_Loop:
    cmp     ecx, MAX_HISTORY
    jge     LH_Done
    mov     eax, slotSerial[ecx*4]
    cmp     eax, 0                      ; 0 = empty slot
    je      LH_Next
    PrintMsg txtRunNo
    call    WriteDec
    PrintMsg txtColon
    mov     eax, slotAlg[ecx*4]
    call    PrintAlgName
    PrintMsg txtProcsOf
    mov     eax, slotN[ecx*4]
    call    WriteDec
    PrintMsg txtProcsEnd
LH_Next:
    inc     ecx
    jmp     LH_Loop
LH_Done:
    call    Crlf
    ret
ListHistory ENDP

; PickRun: list runs and ask for a run number.
; Returns ESI = slot of that run, or -1 if it was replaced.
PickRun PROC USES eax ecx
    call    ListHistory
    INVOKE  ReadRange, ADDR prRunNo, 1, histSerial
    mov     esi, -1
    xor     ecx, ecx
PK_Loop:
    cmp     ecx, MAX_HISTORY
    jge     PK_Gone
    cmp     slotSerial[ecx*4], eax
    je      PK_Found
    inc     ecx
    jmp     PK_Loop
PK_Found:
    mov     esi, ecx
    ret
PK_Gone:
    PrintColored txtRunGone, lightRed
    ret
PickRun ENDP

; CPUPerformanceAnalyzer: analyze the last run, an earlier run or the
; last comparison from stored data.
CPUPerformanceAnalyzer PROC USES eax esi
CP_Loop:
    ShowMenu txtAnTitle, txtAnMenu
    INVOKE  ReadRange, ADDR prChoice, 1, 4
    cmp     eax, 4
    je      CP_Exit
    cmp     eax, 1
    je      CP_Last
    cmp     eax, 2
    je      CP_Prev
    cmp     cmpValid, 0                 ; option 3: last comparison
    jne     CP_Cmp
    PrintMsg txtNoCmpAn
    jmp     CP_Wait
CP_Cmp:
    call    AskThreshold
    mov     eax, CMP_SLOT_A
    call    AnalyzeSlot
    mov     eax, CMP_SLOT_B
    call    AnalyzeSlot
    jmp     CP_Wait
CP_Last:
    cmp     histSerial, 0
    jne     CP_DoLast
    PrintMsg txtNoRunAn
    jmp     CP_Wait
CP_DoLast:
    call    AskThreshold
    mov     eax, lastSlot
    call    AnalyzeSlot
    jmp     CP_Wait
CP_Prev:
    cmp     histSerial, 0
    jne     CP_DoPrev
    PrintMsg txtNoRunAn
    jmp     CP_Wait
CP_DoPrev:
    call    PickRun
    cmp     esi, -1
    je      CP_Wait
    call    AskThreshold
    mov     eax, esi
    call    AnalyzeSlot
CP_Wait:
    call    Crlf
    call    WaitMsg
    jmp     CP_Loop
CP_Exit:
    ret
CPUPerformanceAnalyzer ENDP

; AskThreshold: read the starvation threshold.
AskThreshold PROC USES eax
    INVOKE  ReadRange, ADDR prThreshold, 1, 999
    mov     starveThreshold, eax
    ret
AskThreshold ENDP

; ScanSegments: EAX = slot -> aBusy, aIdle, aSwitch from the stored Gantt.
ScanSegments PROC USES eax ebx ecx edx esi edi
    mov     esi, eax
    mov     ebx, esi
    imul    ebx, MAX_SEGMENTS
    mov     edi, slotSegCnt[esi*4]
    mov     aBusy, 0
    mov     aIdle, 0
    mov     aSwitch, 0
    xor     ecx, ecx
SS_Loop:
    cmp     ecx, edi
    jge     SS_Done
    lea     edx, [ebx+ecx]
    mov     eax, slotGEnd[edx*4]
    sub     eax, slotGStart[edx*4]      ; segment length
    cmp     slotGPid[edx*4], IDLE_PID
    jne     SS_Busy
    add     aIdle, eax
    jmp     SS_Next
SS_Busy:
    add     aBusy, eax
    cmp     ecx, 0                      ; first segment: no predecessor
    je      SS_Next
    mov     eax, slotGPid[edx*4]
    cmp     eax, slotGPid[edx*4 - 4]    ; same process as before?
    je      SS_Next
    cmp     slotGPid[edx*4 - 4], IDLE_PID
    je      SS_Next                     ; coming from IDLE is not a switch
    inc     aSwitch
SS_Next:
    inc     ecx
    jmp     SS_Loop
SS_Done:
    ret
ScanSegments ENDP

; ScanElapsed: EAX = slot -> aElapsed = latest completion time.
ScanElapsed PROC USES eax ecx edx esi
    mov     esi, eax
    mov     aElapsed, 0
    xor     ecx, ecx
SL_Loop:
    cmp     ecx, slotN[esi*4]
    jge     SL_Done
    mov     edx, esi
    imul    edx, MAX_PROCESSES
    add     edx, ecx
    mov     eax, slotCt[edx*4]
    cmp     eax, aElapsed
    jle     SL_Next
    mov     aElapsed, eax
SL_Next:
    inc     ecx
    jmp     SL_Loop
SL_Done:
    ret
ScanElapsed ENDP

; ShowStarvation: EAX = slot. Flag processes whose waiting time is
; above starveThreshold.
ShowStarvation PROC USES eax ebx ecx edx esi
    mov     esi, eax
    mov     starveCount, 0
    PrintColored hdrStarve, yellow
    Border  wStarve, 3
    SetColor yellow
    Bar
    CellStr hPID,    TBL_W
    CellStr hWT,     TBL_W
    CellStr hStatus, 18
    call    Crlf
    SetColor white
    Border  wStarve, 3
    xor     ecx, ecx
SV_Loop:
    cmp     ecx, slotN[esi*4]
    jge     SV_Done
    mov     eax, esi
    call    CalculateMetrics
    Bar
    CellNum mPid, TBL_W
    CellNum mWT,  TBL_W
    mov     eax, mWT
    cmp     eax, starveThreshold
    jle     SV_Ok
    SetColor lightRed
    CellStr txtStarveYes, 18
    SetColor white
    inc     starveCount
    jmp     SV_Next
SV_Ok:
    CellStr txtStarveNo, 18
SV_Next:
    call    Crlf
    inc     ecx
    jmp     SV_Loop
SV_Done:
    Border  wStarve, 3
    PrintMsg txtFlagged
    mov     eax, starveCount
    call    WriteDec
    call    Crlf
    ret
ShowStarvation ENDP

; AnalyzeSlot: EAX = slot. Busy/idle/elapsed time, utilization,
; throughput, context switches and starvation (stored data only).
AnalyzeSlot PROC USES eax ebx edx esi edi
    mov     esi, eax
    SetColor lightGreen
    PrintMsg txtAnHeader
    PrintMsg txtAlgLabel
    mov     eax, slotAlg[esi*4]
    call    PrintAlgName
    call    Crlf
    SetColor white
    mov     eax, esi
    call    ScanSegments
    mov     eax, esi
    call    ScanElapsed
    mov     edx, OFFSET lblBusy
    mov     eax, aBusy
    call    ShowStat
    mov     edx, OFFSET lblIdle
    mov     eax, aIdle
    call    ShowStat
    mov     edx, OFFSET lblElapsed
    mov     eax, aElapsed
    call    ShowStat
    mov     eax, aBusy                  ; check: busy + idle = elapsed
    add     eax, aIdle
    cmp     eax, aElapsed
    jne     AN_Bad
    PrintMsg txtCheckOk
    jmp     AN_Util
AN_Bad:
    PrintMsg txtCheckBad
AN_Util:
    PrintMsg lblUtil                    ; utilization = busy*100 / elapsed
    mov     eax, aBusy
    imul    eax, 100
    mov     edi, aElapsed
    call    Avg100
    call    PrintFixed2
    PrintMsg txtPercent
    PrintMsg lblThru                    ; throughput = processes / elapsed
    mov     eax, slotN[esi*4]
    mov     edi, aElapsed
    call    Avg100
    call    PrintFixed2
    PrintMsg txtPerUnit
    mov     edx, OFFSET lblSwitch
    mov     eax, aSwitch
    call    ShowStat
    mov     eax, esi
    call    ShowStarvation
    PrintMsg txtAnExplain
    ret
AnalyzeSlot ENDP
